// 手写动态数组核心：扩容机制
class DynamicArray {
  constructor() { this.data = []; this.size = 0; this.capacity = 4; }
  push(val) {
    if (this.size === this.capacity) {
      this.capacity *= 1.5; // 扩容 1.5 倍
    }
    this.data[this.size++] = val;
  }
  get(i) { return this.data[i]; } // O(1)
  shift() { // O(n)：所有元素前移
    for (let i = 0; i < this.size - 1; i++)
      this.data[i] = this.data[i + 1];
    this.size--;
  }
}

function reverseList(head) {
  let prev = null, curr = head;
  while (curr) {
    let next = curr.next;  // 先记住下一个
    curr.next = prev;       // 掉头指向前一个
    prev = curr;            // prev 前进
    curr = next;             // curr 前进

}

// 手写哈希表（链地址法）
class HashTable {
  constructor(size = 53) {
    this.table = Array(size).fill(null).map(() => []);
  }
  _hash(key) {
    let h = 0;
    for (const c of key) h = (h * 31 + c.charCodeAt(0)) % this.table.length;
    return h;
  }
  set(key, val) { this.table[this._hash(key)].push([key, val]); }
  get(key) {
    const bucket = this.table[this._hash(key)];
    return bucket.find(([k]) => k === key)?.[1];
  }
}

function traverse(root) {
  if (!root) return;

  // 中序：在这行 console.log(root.val)

// 层序遍历（BFS）
function levelOrder(root) {
  if (!root) return [];
  const q = [root], res = [];
  while (q.length) {
    const level = [];
    const size = q.length;
    for (let i = 0; i < size; i++) {
      const node = q.shift();
      level.push(node.val);
      if (node.left) q.push(node.left);
      if (node.right) q.push(node.right);
    }
    res.push(level);
  }
  return res;
}

// BST 搜索：利用左小右大特性
function searchBST(root, val) {
  while (root) {
    if (val === root.val) return root;
    root = val < root.val ? root.left : root.right; // 小了往左，大了往右
  }
  return null;
}
// 验证 BST：中序遍历必须递增
function isValidBST(root, min = -Infinity, max = Infinity) {
  if (!root) return true;
  if (root.val <= min || root.val >= max) return false;
  return isValidBST(root.left, min, root.val) &&
         isValidBST(root.right, root.val, max);
}

// 最小堆核心操作
class MinHeap {
  constructor() { this.heap = []; }
  insert(val) {
    this.heap.push(val);
    this._siftUp(this.heap.length - 1);
  }
  extractMin() {
    const min = this.heap[0];
    const last = this.heap.pop();
    if (this.heap.length) {
      this.heap[0] = last;
      this._siftDown(0);
    }
    return min;
  }
  _siftUp(i) {
    while (i > 0) {
      const p = Math.floor((i - 1) / 2);
      if (this.heap[p] <= this.heap[i]) break;
      [this.heap[p], this.heap[i]] = [this.heap[i], this.heap[p]];
      i = p;
    }
  }
  _siftDown(i) { /* 对称操作 */ }
}

// 快排：选班长分区法
function quickSort(arr) {
  if (arr.length <= 1) return arr;
  const pivot = arr[0];
  const left = arr.slice(1).filter(x => x < pivot);  // 比班长矮的
  const right = arr.slice(1).filter(x => x >= pivot); // 比班长高的
  return [...quickSort(left), pivot, ...quickSort(right)];
}
// 归并：两组合并法
function mergeSort(arr) {
  if (arr.length <= 1) return arr;
  const mid = Math.floor(arr.length / 2);
  const left = mergeSort(arr.slice(0, mid));
  const right = mergeSort(arr.slice(mid));
  return merge(left, right); // 合并两个有序数组
}

// 经典二分查找
function binarySearch(arr, target) {
  let left = 0, right = arr.length - 1;
  while (left <= right) {
    const mid = left + ((right - left) >> 1); // 防溢出
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

// 对撞指针：两数之和（有序数组）
function twoSum(arr, target) {
  let left = 0, right = arr.length - 1;
  while (left < right) {
    const sum = arr[left] + arr[right];
    if (sum === target) return [left, right];
    if (sum < target) left++;  // 和太小，左指针右移
    else right--;            // 和太大，右指针左移
  }
}
// 快慢指针：判断环形链表
function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;          // 慢的走 1 步
    fast = fast.next.next;     // 快的走 2 步
    if (slow === fast) return true; // 相遇=有环
  }
  return false;
}

// 最长无重复字符子串
function lengthOfLongestSubstring(s) {
  const seen = new Map(); // 窗口内字符记录
  let left = 0, max = 0;
  for (let right = 0; right < s.length; right++) {
    if (seen.has(s[right]))
      left = Math.max(left, seen.get(s[right]) + 1); // 左缩
    seen.set(s[right], right);
    max = Math.max(max, right - left + 1); // 更新最大窗口
  }
  return max;
}

// 回溯模板：全排列
function permute(nums) {
  const result = [];
  function backtrack(path, used) {
    if (path.length === nums.length) {
      result.push([...path]); // 收集结果
      return;
    }
    for (let i = 0; i < nums.length; i++) {
      if (used[i]) continue;
      path.push(nums[i]);     // 做选择
      used[i] = true;
      backtrack(path, used); // 递归
      path.pop();             // 撤销选择（回溯！）
      used[i] = false;
    }
  }
  backtrack([], []);
  return result;
}

// 分发饼干（贪心）
function findContentChildren(g, s) {
  g.sort((a, b) => a - b); // 胃口排序
  s.sort((a, b) => a - b); // 饼干排序
  let i = 0;
  for (const cookie of s) {
    if (i < g.length && cookie >= g[i]) i++; // 最小饼干满足最小胃口
  }
  return i;
}

// DP 三步法：1.定义状态 2.找转移方程 3.确定初始值
// 爬楼梯：dp[i] = dp[i-1] + dp[i-2]
function climbStairs(n) {
  if (n <= 2) return n;
  const dp = [0, 1, 2]; // 3. 初始值
  for (let i = 3; i <= n; i++)
    dp[i] = dp[i - 1] + dp[i - 2]; // 2. 转移方程
  return dp[n]; // 1. dp[i]=爬到第i层的方法数
}
// 空间优化版：滚动变量
function climbStairs2(n) {
  let prev = 1, curr = 2;
  for (let i = 3; i <= n; i++) [prev, curr] = [curr, prev + curr];
  return curr;
}

// 邻接表表示图 + BFS 最短路径
class Graph {
  constructor() { this.adj = new Map(); }
  addEdge(u, v) {
    (this.adj.get(u) || this.adj.set(u, []).get(u)).push(v);
    (this.adj.get(v) || this.adj.set(v, []).get(v)).push(u);
  }
  bfsShortestPath(start, end) {
    const queue = [[start, 0]];
    const visited = new Set([start]);
    while (queue.length) {
      const [node, dist] = queue.shift();
      if (node === end) return dist;
      for (const next of this.adj.get(node) || [])
        if (!visited.has(next)) { visited.add(next); queue.push([next, dist + 1]); }
    }
    return -1;
  }
}

// 前缀和 + 区间查询
class PrefixSum {
  constructor(arr) {
    this.prefix = [0]; // prefix[0] = 0
    for (const num of arr)
      this.prefix.push(this.prefix[this.prefix.length - 1] + num);
  }
  rangeSum(l, r) { // [l, r] 闭区间
    return this.prefix[r + 1] - this.prefix[l];
  }
}

// 每日温度（单调递减栈）
function dailyTemperatures(T) {
  const result = Array(T.length).fill(0);
  const stack = []; // 存索引，栈内温度递减
  for (let i = 0; i < T.length; i++) {
    while (stack.length && T[i] > T[stack[stack.length - 1]]) {
      const top = stack.pop();
      result[top] = i - top; // 找到比栈顶大的了
    }
    stack.push(i);
  }
  return result;
}
